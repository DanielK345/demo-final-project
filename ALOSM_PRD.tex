\section{Executive Summary}
% ==============================================================================

\textbf{AloSM Voice AI Agent} là hệ thống trợ lý hội thoại thông minh cho phép khách hàng tương tác với dịch vụ AloSM bằng tiếng Việt tự nhiên. Sản phẩm đóng vai trò là lớp giao tiếp thông minh giữa khách hàng và hệ thống nghiệp vụ --- không phải một ứng dụng gọi xe mới hay một chatbot thay thế hoàn toàn con người.

\begin{tcolorbox}[visionbox,title={Tầm nhìn sản phẩm}]
  \begin{center}
    {\large\itshape ``Nói một câu, AloSM lo phần còn lại.''}
  \end{center}
  \vspace{0.3em}
  AloSM Voice AI Agent hướng tới trở thành \textbf{cổng giao tiếp bằng hội thoại} cho hệ sinh thái dịch vụ AloSM --- giúp khách hàng tiếp cận dịch vụ một cách tự nhiên, thuận tiện và toàn diện thông qua giọng nói; đồng thời hỗ trợ doanh nghiệp tối ưu quy trình chăm sóc khách hàng và nâng cao hiệu quả vận hành.
\end{tcolorbox}

\subsection*{Sản phẩm là gì}

Người dùng chỉ cần trình bày nhu cầu bằng giọng nói. AI sẽ tiếp nhận yêu cầu, duy trì ngữ cảnh, thu thập và xác nhận thông tin cần thiết, thực hiện nghiệp vụ phù hợp và phản hồi kết quả. Khi yêu cầu vượt ngoài phạm vi xử lý hoặc cần đánh giá của con người, hệ thống sẽ chuyển tiếp cuộc hội thoại đến tổng đài viên theo mô hình Human-in-the-Loop.

\subsection*{Bốn năng lực cốt lõi (Core Capabilities)}

\begin{enumerate}
  \item \textbf{Booking Services:} Thu thập và xác nhận điểm đón, điểm đến, sau đó thực hiện quy trình đặt xe.
  \item \textbf{Information Services:} Tra cứu trạng thái chuyến đi, thông tin dịch vụ, giá tham khảo và các câu hỏi thường gặp.
  \item \textbf{Conversation Management:} Duy trì ngữ cảnh, xử lý hội thoại nhiều lượt và xác nhận thông tin quan trọng.
  \item \textbf{Human Handoff:} Chuyển tiếp cuộc hội thoại đến tổng đài viên khi người dùng yêu cầu trực tiếp, sau hai lần nhận diện hoặc xác nhận thất bại, khi độ tin cậy thấp, hoặc khi xuất hiện tình huống nhạy cảm (khiếu nại, thanh toán, tai nạn). Tổng đài viên nhận được tóm tắt và các thông tin đã xác nhận.
\end{enumerate}

\subsection*{Đối tượng sử dụng chính}
\begin{itemize}
  \item \textbf{Khách hàng (Primary):} Người lớn tuổi chưa quen app, người ít thành thạo công nghệ, người gặp khó khăn khi thao tác trên thiết bị.
  \item \textbf{Khách hàng (Secondary):} Người cần rảnh tay (mang hành lý, bế trẻ nhỏ); người dùng app thành thạo nhưng muốn tiện lợi hơn.
  \item \textbf{Tổng đài viên (Operator):} Tiếp nhận các cuộc gọi phức tạp được chuyển từ AI, xử lý các trường hợp cần chuyên môn.
  \item \textbf{Tài xế:} Nhận và cập nhật trạng thái chuyến từ hệ thống booking.
  \item \textbf{Quản trị viên / AI Operations:} Quản lý cấu hình Agent, nội dung tri thức và chất lượng dịch vụ.
\end{itemize}

\subsection*{Kết quả mong đợi sau MVP}
\begin{itemize}
  \item Chứng minh AI Voice có thể tự động hóa hiệu quả các nghiệp vụ tiêu chuẩn (đặt xe, tra cứu, FAQ).
  \item Thiết lập baseline để đo lường tỉ lệ hoàn thành tác vụ và chất lượng hội thoại.
  \item Xác nhận mức độ chấp nhận của người dùng trước khi mở rộng kênh.
  \item Nền tảng để phát triển thêm tính năng và mở rộng sang điện thoại, app mobile, kiosk.
\end{itemize}

\subsection{Bối cảnh thị trường}

Ngành dịch vụ đang chuyển dịch mạnh từ các giao diện tương tác truyền thống sang các giải pháp Conversational AI. Thay vì yêu cầu người dùng thực hiện nhiều thao tác trên màn hình hoặc tuân theo các quy trình phức tạp, doanh nghiệp ngày càng hướng đến việc cho phép khách hàng giao tiếp bằng ngôn ngữ tự nhiên thông qua giọng nói.

Xu hướng này đặc biệt phù hợp với các doanh nghiệp có quy mô khách hàng lớn và hệ thống chăm sóc khách hàng hoạt động liên tục như ngân hàng, hàng không, bảo hiểm và dịch vụ gọi xe. AI Voice có thể tiếp nhận các yêu cầu phổ biến, tự động hóa quy trình hỗ trợ và giảm tải cho tổng đài viên.

\subsection{Tại sao đây là thời điểm phù hợp? (Why Now?)}

Việt Nam có tỷ lệ sử dụng smartphone cao, nhưng khả năng tiếp cận dịch vụ số giữa các nhóm người dùng vẫn chưa đồng đều --- tạo nên nhu cầu thực sự cho kênh giọng nói:

\begin{itemize}
  \item \textbf{Dân số cao tuổi lớn và tăng nhanh:} Tổng điều tra Dân số 2019 ghi nhận Việt Nam có \textbf{11,41 triệu người từ 60 tuổi trở lên} (11,86\% dân số). Đây là nhóm người dùng ưu tiên kênh giọng nói hơn ứng dụng.
  \item \textbf{Khoảng cách số với người cao tuổi:} Nghiên cứu LSAHV (ERIA và PHAD, 2020) trên mẫu 6.050 người từ 60 tuổi cho thấy chỉ \textbf{12,7\%} có khả năng truy cập Internet --- giảm từ 17,0\% (nhóm 60--69 tuổi) xuống 2,8\% (nhóm từ 80 tuổi trở lên).
  \item \textbf{Nhu cầu gọi xe số đã được xác lập:} Khảo sát Rakuten Insight (2025) ghi nhận 66\% người trả lời từng đặt ô tô, 67\% từng đặt xe máy qua ứng dụng; AloSM là thương hiệu được sử dụng thường xuyên nhất bởi 32\% người trả lời.
  \item \textbf{Quy mô dịch vụ đủ lớn để đánh giá AI Voice:} Hệ thống xử lý hơn \textbf{1 triệu chuyến mỗi ngày} và đã phục vụ hơn \textbf{100 triệu lượt khách} thông qua hotline 1555 và ứng dụng di động.
\end{itemize}

Trong bối cảnh nhu cầu gọi xe số phát triển nhưng khả năng tiếp cận công nghệ chưa đồng đều, đây là thời điểm phù hợp để AloSM bổ thẩm AI Voice như một kênh tương tác thêm cho ứng dụng và tổng đài truyền thống.

\subsection{Các vấn đề hiện tại}
\begin{itemize}
  \item \textbf{Phụ thuộc vào nhân lực tổng đài:} Mỗi cuộc gọi đòi hỏi một tổng đài viên trực tiếp; chi phí vận hành cao và khó mở rộng.
  \item \textbf{Nghẽn cuộc gọi giờ cao điểm:} Số lượng cuộc gọi tăng đột biến vào sáng sớm, chiều tối vượt khả năng phục vụ.
  \item \textbf{Chất lượng phục vụ không đồng nhất:} Mỗi tổng đài viên xử lý khác nhau; thiếu chuẩn hóa.
  \item \textbf{Câu hỏi lặp lại chiếm phần lớn cuộc gọi:} Hỏi giá, hỏi trạng thái, đặt xe đơn giản --- những yêu cầu có thể tự động hóa chiếm tỷ trọng lớn.
  \item \textbf{Người lớn tuổi và người ít quen app gặp rào cản:} Giao diện ứng dụng đòi hỏi kỹ năng thao tác không phải người dùng nào cũng có.
  \item \textbf{Mất ngữ cảnh khi chuyển xử lý:} Khách hàng phải kể lại thông tin khi được chuyển sang người khác.
\end{itemize}

\subsection{Product Principles}

\begin{enumerate}
  \item \textbf{Voice First.} Giọng nói là phương thức tương tác chính, cho phép người dùng tiếp cận dịch vụ bằng hội thoại tự nhiên. Luôn có phương án nhập bằng bàn phím khi microphone không hoạt động.
  \item \textbf{Human-Centered AI.} AI xử lý các nghiệp vụ tiêu chuẩn; con người tiếp nhận những tình huống cần phán đoán hoặc hỗ trợ chuyên sâu. AI hỗ trợ con người, không thay thế con người.
  \item \textbf{Reliable Automation.} Hệ thống chỉ tự động hóa những nghiệp vụ có quy trình rõ ràng, có thể kiểm soát và xác nhận. Các tình huống không rõ quy trình (khiếu nại, tranh chấp, tai nạn) phải được chuyển ngay sang người thật.
  \item \textbf{Continuous Conversation.} AI duy trì ngữ cảnh trong nhiều lượt trao đổi, ghi nhớ thông tin trong phiên và phản hồi nhất quán. Người dùng không phải nhắc lại những gì đã nói trong cùng một cuộc gọi.
  \item \textbf{Xác nhận trước hành động quan trọng.} Mọi booking, hủy chuyến, hoặc thay đổi thông tin quan trọng đều phải có bước xác nhận rõ ràng từ khách hàng trước khi thực hiện.
  \item \textbf{Không tự suy đoán dữ liệu nghiệp vụ.} Khi thông tin còn thiếu hoặc mơ hồ, AI hỏi lại thay vì tự điền giá trị. Không tự chọn địa chỉ khi có nhiều kết quả tương đương.
  \item \textbf{Chuyển người thật khi độ tin cậy thấp.} Sau hai lần nhận diện hoặc xác nhận thất bại, hoặc khi phát hiện tình huống nhạy cảm, hệ thống chuyển sang tổng đài viên với đầy đủ ngữ cảnh.
  \item \textbf{Minh bạch về trạng thái xử lý.} Khách hàng luôn biết hệ thống đang lắng nghe, đang xử lý hay đang nói. Không để khách hàng chờ trong im lặng.
  \item \textbf{Bảo vệ dữ liệu cá nhân.} Không đọc toàn bộ số điện thoại qua TTS. Che dữ liệu PII trong giao diện theo vai trò. Ghi âm chỉ khi có consent.
  \item \textbf{Có phương án fallback khi voice không hoạt động.} Mất kết nối, lỗi microphone, hoặc ASR thất bại phải có xử lý dự phòng rõ ràng.
\end{enumerate}

\subsection{Core Values}

\begin{description}[leftmargin=3em,style=nextline]
  \item[\textcolor{AloBlue}{\textbf{Accessibility}}] Giúp nhiều nhóm khách hàng tiếp cận dịch vụ dễ dàng hơn thông qua giọng nói --- đặc biệt người lớn tuổi, người ít thành thạo công nghệ và người gặp khó khăn khi thao tác trên thiết bị.
  \item[\textcolor{AloBlue}{\textbf{Convenience}}] Đơn giản hóa quá trình sử dụng dịch vụ bằng hội thoại tự nhiên --- người dùng chỉ cần trình bày nhu cầu, hệ thống lo phần còn lại.
  \item[\textcolor{AloBlue}{\textbf{Operational Efficiency}}] Tự động hóa các nghiệp vụ chuẩn hóa nhằm giảm tải tổng đài, tối ưu nguồn lực và chuẩn hóa chất lượng phục vụ.
  \item[\textcolor{AloBlue}{\textbf{Human-Centered AI}}] AI hỗ trợ con người thay vì thay thế con người. Các tình huống cần đánh giá, xử lý linh hoạt hoặc ra quyết định của con người luôn được chuyển đến tổng đài viên.
\end{description}

\section{Mục tiêu sản phẩm}

\subsection{Mục tiêu người dùng}
\begin{itemize}
  \item Khách hàng có thể đặt xe hoàn toàn bằng giọng nói, không cần nhập liệu thủ công.
  \item Khách hàng không phải lặp lại thông tin đã cung cấp trong cùng một cuộc gọi.
  \item Khách hàng nhận được phản hồi rõ ràng về giá, địa chỉ và trạng thái booking.
  \item Khách hàng có thể sửa thông tin sai bất kỳ lúc nào trước khi xác nhận.
  \item Khách hàng có thể yêu cầu gặp tổng đài viên bất kỳ lúc nào mà không phải cúp cuộc gọi.
\end{itemize}

\subsection{Mục tiêu kinh doanh}
\begin{itemize}
  \item Chứng minh AI Voice có thể tự động hóa hiệu quả các nghiệp vụ tiêu chuẩn --- đặt xe, tra cứu trạng thái, giải đáp FAQ.
  \item Thiết lập baseline đo lường để đánh giá tác động giảm tải tổng đài sau pilot.
  \item Tăng tổng số cuộc gọi có thể phục vụ đồng thời mà không tăng tương ứng nhân sự.
  \item Chuẩn hóa chất lượng thông tin và quy trình phục vụ qua mỗi cuộc gọi.
  \item Đo lường và cải thiện tỉ lệ hoàn thành đặt xe qua kênh giọng nói.
\end{itemize}

\subsection{Mục tiêu vận hành}
\begin{itemize}
  \item Cung cấp công cụ theo dõi chất lượng từng cuộc gọi và phiên hội thoại.
  \item Cho phép phân tích nguyên nhân thất bại để cải thiện Agent.
  \item Quản lý tri thức tập trung, có phiên bản và kiểm soát xuất bản.
  \item Cung cấp báo cáo đủ để đưa ra quyết định sản phẩm dựa trên dữ liệu.
\end{itemize}

\section{Product Scope}
% ==============================================================================

\subsection{MVP --- In Scope}
\begin{itemize}
  \item Tiếp nhận và phản hồi hội thoại bằng giọng nói tiếng Việt.
  \item Hiểu ý định và duy trì ngữ cảnh trong nhiều lượt trao đổi.
  \item \textbf{Đặt xe:} Thu thập điểm đón, điểm đến, loại xe; xác nhận thông tin; tạo booking.
  \item \textbf{Tra cứu trạng thái chuyến đi:} Khách hỏi trạng thái chuyến, AI truy vấn qua tool.
  \item \textbf{Giải đáp thông tin dịch vụ và FAQ:} Giá cước, chính sách, dịch vụ.
  \item Xác nhận thông tin quan trọng trước khi thực hiện nghiệp vụ.
  \item Xử lý địa chỉ mơ hồ: hiển thị danh sách, hỏi khách chọn --- không tự chọn.
  \item Báo giá dự kiến trước khi xác nhận booking.
  \item Lưu lịch sử hội thoại trong phiên.
  \item Chuyển tiếp tổng đài viên (HITL) khi cần.
  \item Operator Queue và Support Console cơ bản.
  \item Lịch sử cuộc gọi và lịch sử chuyến.
  \item Knowledge Base cơ bản (upload tài liệu, tìm kiếm ngữ nghĩa).
  \item Dashboard quản trị cơ bản với KPI chính.
  \item Quản lý giọng trợ lý (chọn voice profile).
  \item Tài xế nhận và xác nhận chuyến từ hệ thống.
\end{itemize}

\subsection{Out of Scope (MVP)}
\begin{itemize}
  \item Thanh toán, hoàn tiền và xử lý giao dịch tài chính.
  \item Khiếu nại, tranh chấp và đánh giá trách nhiệm.
  \item Điều phối tài xế hoặc can thiệp vào thuật toán ghép chuyến.
  \item Tai nạn, tình huống khẩn cấp và các trường hợp đặc biệt.
\end{itemize}

\begin{tcolorbox}[notebox]
  \textbf{Lưu ý:} AloSM Voice AI Agent không hướng đến việc thay thế tổng đài viên. MVP chỉ chứng minh rằng AI có thể tự động hóa hiệu quả các nghiệp vụ tiêu chuẩn, giúp con người tập trung vào những tình huống cần chuyên môn và khả năng đánh giá.
\end{tcolorbox}

\subsection{Post-MVP}
\begin{itemize}
  \item \textbf{Mở rộng nghiệp vụ:} Hủy chuyến đầy đủ, báo mất đồ, tiếp nhận phản ánh, hỗ trợ khuyến mãi và khách hàng doanh nghiệp.
  \item \textbf{Kênh telephony:} Tích hợp đường dây điện thoại thật (SIP/VoIP); triển khai kiosk tự phục vụ.
  \item \textbf{Mobile native app:} Tích hợp Voice Agent vào ứng dụng di động AloSM (iOS/Android).
  \item \textbf{Cá nhân hóa:} Ghi nhớ lịch sử tương tác, địa chỉ thường dùng, sở thích giọng nói.
  \item \textbf{Mở rộng kênh:} Website, Zalo OA, Messenger; đa ngôn ngữ (tiếng Anh, v.v.).
  \item Theo dõi xe thời gian thực. \quad Kết nối CRM. \quad Kiến trúc multi-agent.
  \item A/B testing cho cấu hình Agent. \quad Evaluation Lab tự động.
\end{itemize}

\subsection{Future Vision}
\begin{itemize}
  \item Nền tảng Voice AI phục vụ nhiều dịch vụ trong hệ sinh thái AloSM.
  \item \textbf{Lớp giao tiếp bằng hội thoại thống nhất} (Unified Conversational Layer) cho toàn bộ hệ sinh thái.
  \item Cá nhân hóa trải nghiệm, lịch sử tương tác xuyên phiên, đa quốc gia.
  \item Phân tích cuộc gọi nâng cao bằng AI; cải thiện liên tục mô hình qua dữ liệu thực tế.
\end{itemize}

\section{Non-functional Requirements}
% ==============================================================================

\subsection{Usability}
\begin{itemize}
  \item Người dùng phổ thông có thể bắt đầu cuộc gọi trong tối đa ba thao tác từ trang chủ.
  \item Ba trạng thái cuộc gọi phân biệt rõ bằng cả màu sắc, icon và âm thanh.
  \item Các hành động chính (Gọi AI, Xác nhận, Hủy, Gặp người thật) phải dễ tìm.
  \item Giao diện hoạt động đúng trên màn hình desktop ($\geq$1024px) và mobile ($\geq$375px).
\end{itemize}

\subsection{Accessibility}
\begin{itemize}
  \item Cỡ chữ tối thiểu 16px cho nội dung chính; tiêu đề không nhỏ hơn 14px.
  \item Tỉ lệ tương phản màu sắc tối thiểu 4.5:1.
  \item Nút hành động tối thiểu 44$\times$44px.
  \item Có text fallback cho mọi trạng thái voice.
  \item Hỗ trợ người lớn tuổi: cỡ chữ có thể tăng, tốc độ TTS có thể giảm.
\end{itemize}

\subsection{Performance (mức sản phẩm)}
\begin{itemize}
  \item Hiển thị chỉ báo ``đang xử lý'' trong vòng 500ms kể từ khi người dùng dừng nói.
  \item Giao diện không được đóng băng trong quá trình AI xử lý.
  \item Có cơ chế timeout và retry rõ ràng cho kết nối chậm.
\end{itemize}

\subsection{Reliability}
\begin{itemize}
  \item Không tạo booking trùng trong cùng phiên, dù người dùng kích hoạt nhiều lần.
  \item Không mất context của cuộc gọi khi chuyển sang tổng đài viên.
  \item Khi kết nối gián đoạn và khách kết nối lại, phiên và thông tin được khôi phục.
  \item Lỗi tool AI được xử lý có kiểm soát; lỗi nội bộ không hiển thị ra người dùng.
\end{itemize}

\subsection{Privacy và Bảo mật}
\begin{itemize}
  \item Ghi âm cuộc gọi chỉ khi có consent của khách hàng.
  \item Dữ liệu PII (số CMND, số tài khoản) không được hiển thị đầy đủ.
  \item Kiểm soát truy cập theo vai trò.
  \item Hỗ trợ yêu cầu xóa dữ liệu cá nhân; có audit record về hành động xóa.
\end{itemize}

\subsection{Scalability}
\begin{itemize}
  \item Kiến trúc sản phẩm hỗ trợ thêm kênh giao tiếp mới mà không phải viết lại luồng cốt lõi.
  \item Hỗ trợ thêm ngôn ngữ mà không thay đổi cấu trúc Agent core.
  \item Có thể thêm nghiệp vụ mới thông qua cấu hình Agent mới.
\end{itemize}

\subsection{Giả định (Assumptions)}
\begin{itemize}
  \item Người dùng có thiết bị với microphone hoạt động khi sử dụng kênh web.
  \item Kết nối internet đủ ổn định để truyền âm thanh (tối thiểu 1Mbps).
  \item Có dữ liệu địa chỉ Việt Nam đủ phủ ít nhất các thành phố trong phạm vi pilot.
  \item Có dịch vụ booking nghiệp vụ (hoặc sandbox) để tích hợp trong MVP.
  \item Có tổng đài viên trực ca để tiếp nhận HITL ít nhất trong giờ hành chính.
  \item Có tài liệu chính sách, FAQ và giá cước đã được phê duyệt để nạp vào Knowledge Base.
  \item Có giọng TTS hợp lệ theo giấy phép để sử dụng trong sản phẩm.
  \item Khách hàng nói tiếng Việt --- MVP không cam kết hỗ trợ tiếng Anh.
\end{itemize}

\subsection{Phụ thuộc (Dependencies)}
\begin{itemize}
  \item \textbf{Hệ thống tài khoản:} API xác thực, quản lý phiên và phân quyền.
  \item \textbf{Dịch vụ bản đồ / địa chỉ:} API tra cứu và gợi ý địa chỉ tiếng Việt.
  \item \textbf{Booking service:} API tạo, sửa, hủy, tra cứu booking.
  \item \textbf{Hạ tầng telephony:} Cần nhà cung cấp SIP/VoIP nếu mở rộng sang điện thoại (post-MVP).
  \item \textbf{Knowledge Base pipeline:} Hệ thống embedding, vector store và retrieval.
  \item \textbf{Đội vận hành tổng đài:} Ít nhất N operator để hỗ trợ HITL trong pilot.
  \item \textbf{Đội bảo mật và DPO:} Phê duyệt chính sách consent, lưu trữ và xóa dữ liệu.
  \item \textbf{Nhà cung cấp TTS/STT:} Giấy phép, SLA và khả năng tùy biến giọng tiếng Việt.
  \item \textbf{Baseline vận hành:} Dữ liệu lưu lượng, cơ cấu yêu cầu và chi phí tổng đài hiện tại của AloSM --- cần thiết trước khi đánh giá tác động MVP.
\end{itemize}
